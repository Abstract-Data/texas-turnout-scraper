## P1-Q1. Extract `Source` enum to typed members (RF-SMELL-003)

**Files:** `src/texas_turnout_scraper/enums.py`, `src/texas_turnout_scraper/cli.py` (45+ touch
points)

Current state: `Source.CIVIX.value` and the string literals `"civix"` / `"legacy"` are passed
through 45+ functions as `source: str`. This is primitive obsession — every caller has to
know the magic strings.

The enum already exists at `enums.py:Source` (verify with grep). The fix is **stop calling
`.value` everywhere** and accept the enum directly.

Target signature:

```python
# enums.py — verify or add
from enum import StrEnum

class Source(StrEnum):
    CIVIX = "civix"
    LEGACY = "legacy"

    @property
    def data_root(self) -> Path:
        """Default data root for this source: data/elections/{civix|legacy}/"""
        return Path("data/elections") / self.value
```

Sweep: replace `source: str` parameters across `cli.py`, `writer.py`, `gap_analysis.py`,
`terminal_report.py` with `source: Source`. At Typer boundaries, accept the string and
coerce at the entry point only.

**Why `StrEnum`:** it stringifies cleanly for f-strings (`f"{source}"` → `"civix"`) and is
JSON-serializable as a string, so existing `roster_*.csv` and `audit_*.json` outputs don't
change shape. Pydantic v2 auto-coerces `str → Source` on `model_validate_json`.

**Verify:** `grep -rn 'source: str' src/` should drop to zero (except at Typer command
boundaries where the Typer arg type is explicit).

```python
# Before
```

```python
# After
def _emit_roster_rows(records: list[VoterRecord], sink: TextIO) -> None:
    writer = csv.writer(sink)
    writer.writerow(ROSTER_CSV_COLUMNS)
    for r in records:

def write_roster_csv(records: list[VoterRecord], path: Path) -> None:
    with path.open("w", newline="", encoding="utf-8") as f:
        _emit_roster_rows(records, f)

    buf = io.StringIO()
```

**File:** `src/texas_turnout_scraper/voterfile.py:399-403` (and any other `duckdb.connect()`
call that doesn't use context manager)

DuckDB 0.10+ supports `with duckdb.connect() as conn:`. Verify via Context7
(`/duckdb/duckdb-python` → `connect` section).

```python

conn = duckdb.connect()
rows = conn.execute(sql, [roster_vuids]).fetchall()

# After
with duckdb.connect() as conn:
```

```python
def _fetch_all_impl(
    source: RosterSource,
    election_id: str,
    *,
    output_dir: Path,
    pace: float,
    dry_run: bool,
    write_audit: bool,
    index_path: Path,
    county_ids: list[str] | None = None,
) -> int:
    """Fetch all EV-date rosters for one election using the given source.

    Returns the typer exit code (0 success, 1 partial failure with --strict-failures,
    2 dry-run-only). Does NOT call typer.Exit — the caller does.

    Side effects:
      - Writes {output_dir}/{election_id}/roster_ev_{election_id}.csv
      - Writes {output_dir}/{election_id}/turnout_ev_{date}.csv per EV date
      - Optionally writes {output_dir}/{election_id}/audit_ev_{election_id}.json
      - Updates {index_path} with the new election entry
    """
```

Implementation outline (~60 LOC):

```python
def _fetch_all_impl(source, election_id, *, output_dir, pace, dry_run, write_audit, index_path, county_ids=None):
    elections = source.list_elections()
    election = _resolve_election(elections, election_id, source=source.source_prefix)
    if election is None:
        typer.echo(f"Election {election_id} not found in {source.source_prefix}", err=True)
        return 1

    ev_dates = _election_ev_dates(election)
    typer.echo(f"Election: {_election_label(election)}")
    typer.echo(f"EV dates: {len(ev_dates)}")

    if dry_run:
        for ev in ev_dates:
            typer.echo(f"[{ev.date}] Would fetch counties (dry-run skips fetches)")
        return 0

    rosters, failures = source.fetch_election_rosters(
        election_id=election_id,
        pace_seconds=pace,
        county_ids=county_ids,
        on_progress=lambda msg: typer.echo(msg),
    )

    roster_csv = _roster_csv_path(source.source_prefix, election_id, output_dir)
    accumulate_roster(rosters, roster_csv)

    if write_audit:
        records = read_roster_csv(roster_csv)
        report = audit_records(records, turnout=_load_stored_turnout(output_dir, election_id))
        _write_audit_json(report, _audit_path(source.source_prefix, election_id, output_dir))

    _update_election_index(
        index_path=index_path,
        source=source.source_prefix,
        election=election,
        roster_csv=roster_csv,
    )

    if failures:
        _print_partial_failure_summary(failures)
        return 1
    return 0
```

```python
def _build_index_entries(
    elections: Sequence[ElectionSummary],
    *,
    source: Source,
    output_dir: Path,
    election_date_fn: Callable[[object], dt.date],
) -> list[IndexEntry]:
    entries: list[IndexEntry] = []
    for e in elections:
        roster_csv = _roster_csv_path(source, e.source_election_id, output_dir)
        entry = IndexEntry(

            source_election_id=e.source_election_id,
```

### Step 1 — define a small dataclass for I/O paths

`cli/_voterfile_runner.py`:

```python
from dataclasses import dataclass

@dataclass(frozen=True, slots=True)
class VoterfilePaths:
    """Resolved I/O paths for one voterfile match run."""
    roster_csv: Path
    voterfile: Path
    out_dir: Path
    matched_csv: Path
    report_json: Path
    sidecar: Path

    @classmethod
    def resolve(
        cls,
        roster_csv: Path,
        voterfile: Path,
        out_dir: Path | None,
        mapping_file: Path | None,
    ) -> VoterfilePaths:
        from ..voterfile import sidecar_path_for
        resolved_out = out_dir or roster_csv.parent
        stem = roster_csv.stem
        return cls(
            roster_csv=roster_csv,
            voterfile=voterfile,
            out_dir=resolved_out,
            matched_csv=resolved_out / f"matched_{stem}.csv",
            report_json=resolved_out / f"match_report_{stem}.json",
            sidecar=mapping_file or sidecar_path_for(voterfile),
        )
```

Use the `Python Dataclasses — Advanced Patterns` playbook from Notion (DEV-ENV-INDEX) to
confirm conventions: `frozen=True, slots=True`, factory method for resolution, no
side effects in `__post_init__`.

### Step 2 — define the orchestrator

`cli/_voterfile_runner.py:_voterfile_match_impl` (target: 25-35 LOC):

```python
def _voterfile_match_impl(
    *,
    roster_csv: Path,
    voterfile: Path,
    output_dir: Path | None,
    mapping_file: Path | None,
    save_mapping: bool,
    no_interactive: bool,
    redetect: bool,
    count_voterfile: bool,
    report_only: bool,
    gap_report: bool,
    gap_turnout_source: str,
) -> int:
    """Orchestrate a voterfile match run. Returns typer exit code."""
    paths = VoterfilePaths.resolve(roster_csv, voterfile, output_dir, mapping_file)
    _exit_if_missing(paths)

    _print_match_header(paths)
    roster = _load_roster(paths.roster_csv)

    mapping, confidence = _resolve_mapping(
        paths=paths,
        save_mapping=save_mapping,
        no_interactive=no_interactive,
        redetect=redetect,
    )
    _print_mapping_summary(mapping, confidence)

    enriched, report = match_voterfile_to_roster(
        roster_records=roster,
        voterfile_path=paths.voterfile,
        mapping=mapping,
        count_voterfile=count_voterfile,
    )

    written = _write_match_outputs(
        paths=paths,
        enriched=enriched,
        report=report,
        report_only=report_only,
    )

    if gap_report:
        _attach_gap_report(report, source=gap_turnout_source, roster_csv=paths.roster_csv)

    _print_match_summary(report, written)
    return 0
```

## P4-2. DuckDB CSV path embedding — defense in depth (Code Review P1-SEC-002)

**File:** `src/texas_turnout_scraper/voterfile.py:378-391` (the `f"… '{vf_path_str}' …"`
block)

Even with `vf_path_str = str(voterfile_path).replace("'", "''")`, embedding a user-controlled
path directly into a DuckDB SQL string is a higher-risk pattern than necessary. Switch to
DuckDB parameterized path:

```python
# Before
sql = f"""
    SELECT {quoted_cols}
    FROM read_csv(
        '{vf_path_str}',
        header = true,
        all_varchar = true,
        encoding = 'utf-8'
    )
    WHERE ...
"""
with duckdb.connect() as conn:
    rows = conn.execute(sql, [roster_vuids]).fetchall()

# After (DuckDB supports parameterized paths via `?`)
sql = f"""
    SELECT {quoted_cols}
    FROM read_csv(
        ?,
        header = true,
        all_varchar = true,
        encoding = 'utf-8'
    )
    WHERE right(lpad(trim(cast({quoted_vuid_col} as varchar)), 10, '0'), 10)
          IN (SELECT unnest(?))
"""
with duckdb.connect() as conn:
    cursor = conn.execute(sql, [str(voterfile_path), roster_vuids])
    col_names = [desc[0] for desc in cursor.description]
    rows = cursor.fetchall()
```

Verify the parameterized path syntax via Context7 (`/duckdb/duckdb-python` → `read_csv` +
prepared statements). If parameterized paths are not supported in the installed DuckDB
version, fall back to using `pathlib.Path.resolve()` + an allowlist check that confirms the
path is under a configured `--voterfile-dir`. Document either choice in
`docs/RUNBOOK.md`.

This is a NEVER DO violation per AGENTS.md ("Never buffer the bulk ZIP in memory — stream to
disk"). Use httpx response streaming:

```python
# Before (approximate; verify lines)
response = client.get(url)
zip_bytes = response.content  # multi-MB buffer
with zipfile.ZipFile(io.BytesIO(zip_bytes)) as zf:
    ...

# After
import tempfile
with tempfile.TemporaryDirectory() as tmp:
    zip_path = Path(tmp) / "statewide.zip"
    with client.stream("GET", url) as response:
        response.raise_for_status()
        with zip_path.open("wb") as f:
            for chunk in response.iter_bytes(chunk_size=65536):
                f.write(chunk)
    with zipfile.ZipFile(zip_path) as zf:
        ...
```

Verify httpx stream API via Context7 (`/encode/httpx` → `stream`).

**Verify:** add `tests/unit/test_civix_strategy_b.py::test_fetch_statewide_streams_to_disk`
that mocks an httpx stream and asserts the request was opened with `client.stream(...)`, not
`client.get(...)`.
